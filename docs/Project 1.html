get_variable_values <- function(year) {
  if (length(year) != 1 || !year %in% 2021:2024) {
    stop("year must be 2021, 2022, 2023, or 2024")
  }

  url <- paste0(
    "https://api.census.gov/data/",
    year,
    "/acs/acs1/pums/variables.json"
  )

  response <- GET(url)
  stop_for_status(response)

  json_text <- content(response, as = "text", encoding = "UTF-8")
  variables <- fromJSON(json_text, simplifyVector = FALSE)$variables

  state_variable <- if (year <= 2022) "ST" else "STATE"

  keep <- c(
    "AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP",
    "PWGTP", "FER", "HHL", "SCH", "SCHL", "SEX",
    state_variable, "REGION", "DIVISION"
  )

  missing_variables <- setdiff(keep, names(variables))
  if (length(missing_variables) > 0) {
    stop(
      "Variables missing from the ", year, " JSON: ",
      paste(missing_variables, collapse = ", ")
    )
  }

  #Keep each requested variable's value definitions.
  lapply(variables[keep], function(variable) variable$values)
}

The API returns an array: its first row contains column names, and the remaining rows contain observations.

response_to_tibble <- function(response) {
  stop_for_status(response)

  api_data <- fromJSON(
    content(response, as = "text", encoding = "UTF-8"),
    simplifyMatrix = TRUE
  )

  if (!is.matrix(api_data) || nrow(api_data) < 2) {
    stop("The API did not return any data rows.")
  }

  result <- as_tibble(
    as.data.frame(api_data[-1, , drop = FALSE]),
    .name_repair = "minimal"
  )

  names(result) <- api_data[1, ]
  result
}

3. Helpers for labels and commute times

The time variables contain codes for intervals, so we look up each code’s description in the saved .rds file and calculate the interval’s midpoint.

get_item_labels <- function(variable_values) {
  items <- variable_values$item

  if (is.null(items)) {
    return(setNames(character(), character()))
  }

  unlist(items, use.names = TRUE)
}

#Return seconds
time_label_to_seconds <- function(label) {
  pattern <- paste0(
    "(?i)^(\\d{1,2}):(\\d{2})\\s*([ap])\\.m\\.\\s+to\\s+",
    "(\\d{1,2}):(\\d{2})\\s*([ap])\\.m\\."
  )

  parts <- regmatches(label, regexec(pattern, label, perl = TRUE))[[1]]

  if (length(parts) != 7) {
    return(NA_real_)
  }

  to_minutes <- function(hour, minute, am_pm) {
    hour <- as.integer(hour) %% 12
    if (tolower(am_pm) == "p") hour <- hour + 12

    hour * 60 + as.integer(minute)
  }

  start <- to_minutes(parts[2], parts[3], parts[4])
  end <- to_minutes(parts[5], parts[6], parts[7])

  #An interval that crosses midnight.
  if (end < start) end <- end + 24 * 60

  midpoint_minutes <- ((start + end) / 2) %% (24 * 60)
  midpoint_minutes * 60
}

parse_time_variable <- function(codes, variable_values) {
  labels <- get_item_labels(variable_values)

  midpoint_lookup <- vapply(
    labels,
    time_label_to_seconds,
    numeric(1)
  )

  seconds <- unname(midpoint_lookup[as.character(codes)])
  hms::as_hms(seconds)
}

parse_numeric_variable <- function(codes, variable_values) {
  numbers <- suppressWarnings(as.numeric(codes))
  labels <- get_item_labels(variable_values)

  missing_codes <- names(labels)[
    grepl("^N/A", labels, ignore.case = TRUE)
  ]

  numbers[as.character(codes) %in% missing_codes] <- NA_real_
  numbers
}

parse_categorical_variable <- function(codes, variable_values) {
  labels <- get_item_labels(variable_values)

  if (length(labels) == 0) {
    return(factor(codes))
  }

  #Put numbered codes in numerical order rather than JSON order.
  label_order <- order(
    suppressWarnings(as.numeric(names(labels))),
    names(labels),
    na.last = TRUE
  )
  labels <- labels[label_order]

  factor(
    as.character(codes),
    levels = names(labels),
    labels = unname(labels)
  )
}

4. Write the single-year API function

get_census_data <- function(
    year = 2024,
    numeric_vars = c("AGEP", "PWGTP"),
    categorical_vars = "SEX",
    geography = "State",
    geo_value = NULL,
    dictionary_file = "data/variable_values.rds",
    key = Sys.getenv("CENSUS_API_KEY")
) {
  allowed_years <- 2021:2024
  allowed_numeric <- c(
    "AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP"
  )
  allowed_categorical <- c("FER", "HHL", "SCH", "SCHL", "SEX")
  time_vars <- c("JWAP", "JWDP")

  # Validate the year.
  if (length(year) != 1 || is.na(year) ||
      !year %in% allowed_years) {
    stop("year must be one of 2021, 2022, 2023, or 2024.")
  }

  #Validate numeric variables
  if (!is.character(numeric_vars) ||
      anyNA(numeric_vars) ||
      anyDuplicated(numeric_vars) ||
      !"PWGTP" %in% numeric_vars ||
      !any(numeric_vars %in% allowed_numeric) ||
      any(!numeric_vars %in% c(allowed_numeric, "PWGTP"))) {
    stop(
      "numeric_vars must include PWGTP and at least one ",
      "allowed numeric variable."
    )
  }

  #Validate categorical variables
  if (!is.character(categorical_vars) ||
      length(categorical_vars) == 0 ||
      anyNA(categorical_vars) ||
      anyDuplicated(categorical_vars) ||
      any(!categorical_vars %in% allowed_categorical)) {
    stop(
      "categorical_vars must include at least one of: ",
      paste(allowed_categorical, collapse = ", ")
    )
  }

  #Only one geography level
  if (length(geography) != 1 ||
      is.na(geography) ||
      !geography %in% c("Region", "Division", "State")) {
    stop("geography must be exactly one of Region, Division, or State.")
  }

  if (!nzchar(key)) {
    stop("Set CENSUS_API_KEY in .Renviron or provide key = ...")
  }

  # Load the values saved in the previous part.
  all_values <- readRDS(dictionary_file)
  year_values <- all_values[[as.character(year)]]

  if (is.null(year_values)) {
    stop("The .rds file does not contain values for this year.")
  }

  #State variable changed names after 2022
  geography_variable <- switch(
    geography,
    Region = "REGION",
    Division = "DIVISION",
    State = if (year <= 2022) "ST" else "STATE"
  )

  api_geography <- tolower(geography)

  #Defaults: North Carolina, South Atlantic division, South region.
  if (is.null(geo_value)) {
    geo_value <- switch(
      geography,
      State = "37",
      Division = "5",
      Region = "3"
    )
  }

  if (length(geo_value) != 1 || is.na(geo_value)) {
    stop("geo_value must be one code or 'all'.")
  }

  geo_value <- as.character(geo_value)

  if (tolower(geo_value) == "all") {
    geo_value <- "*"
  } else {
    allowed_codes <- names(
      get_item_labels(year_values[[geography_variable]])
    )

    #Exclude Puerto Rico.
    if (geography == "State") {
      allowed_codes <- setdiff(allowed_codes, "72")
    } else {
      allowed_codes <- setdiff(allowed_codes, "0")
    }

    if (!geo_value %in% allowed_codes) {
      stop(
        "Invalid ", geography, " code. Check the codes in ",
        dictionary_file, "."
      )
    }
  }

  requested_vars <- unique(c(numeric_vars, categorical_vars))

  url <- paste0(
    "https://api.census.gov/data/",
    year,
    "/acs/acs1/pums"
  )

  response <- GET(
    url,
    query = list(
      get = paste(requested_vars, collapse = ","),
      `for` = paste0(api_geography, ":", geo_value),
      key = key
    )
  )

  census_tibble <- response_to_tibble(response)

  #Convert the weight and other numeric variables
  census_tibble$PWGTP <- as.numeric(census_tibble$PWGTP)

  for (variable in setdiff(numeric_vars, "PWGTP")) {
    if (variable %in% time_vars) {
      census_tibble[[variable]] <- parse_time_variable(
        census_tibble[[variable]],
        year_values[[variable]]
      )
    } else {
      census_tibble[[variable]] <- parse_numeric_variable(
        census_tibble[[variable]],
        year_values[[variable]]
      )
    }
  }

  #Convert categorical codes to labeled factors
  for (variable in categorical_vars) {
    census_tibble[[variable]] <- parse_categorical_variable(
      census_tibble[[variable]],
      year_values[[variable]]
    )
  }

  #API adds a lowercase geography-code column
  #Add a readable factor while retaining that code column
  census_tibble$geography_label <- parse_categorical_variable(
    census_tibble[[api_geography]],
    year_values[[geography_variable]]
  )

  class(census_tibble) <- c("census", class(census_tibble))
  census_tibble
}

I combined the earlier helpers into get_census_data(), which queries one year of PUMS data and returns a formatted tibble. The function defaults to 2024, age and person weight as numeric variables, sex as a categorical variable, and North Carolina as the state. The user can choose other permitted variables and request one state, division, or region, including all values within that geography level.

Before sending the request, the function checks that the year, variables, and geography are valid. It requires PWGTP, at least one other numeric variable, and at least one categorical variable. After receiving the data, it converts measurements to numeric values, commute-time codes to interval midpoints, and categorical codes to labeled factors using the selected year’s saved variable values. Finally, it adds a readable geography label and the census class so the data can be used with the custom summary and plotting functions later in the project.

get_census_years <- function(years = 2021:2024, ...) {
  if (!is.numeric(years) ||
      length(years) < 2 ||
      anyNA(years) ||
      anyDuplicated(years) ||
      any(!years %in% 2021:2024)) {
    stop("Provide at least two distinct years from 2021 through 2024.")
  }

  year_data <- lapply(years, function(current_year) {
    one_year <- get_census_data(
      year = current_year,
      ...
    )

    one_year$year <- factor(
      as.character(current_year),
      levels = as.character(years)
    )

    one_year
  })

  combined <- dplyr::bind_rows(year_data)
  class(combined) <- c("census", setdiff(class(combined), "census"))

  combined
}

Numeric summaries

summary.census <- function(object,
                           numeric_vars = NULL,
                           categorical_vars = NULL) {
  
  if (!"PWGTP" %in% names(object)) {
    stop("The census tibble must contain PWGTP.")
  }
  
  time_vars <- c("JWAP", "JWDP", "JWMNP")
  
  #Select all numeric variables except PWGTP, year, and the time variables
  if (is.null(numeric_vars)) {
    numeric_vars <- names(object)[
      vapply(object, is.numeric, logical(1))
    ]
    
    numeric_vars <- setdiff(
      numeric_vars,
      c("PWGTP", "year", time_vars)
    )
  }
  
  #Select all factor variables
  if (is.null(categorical_vars)) {
    categorical_vars <- names(object)[
      vapply(object, is.factor, logical(1))
    ]
    
    #Treat year as categorical
    if ("year" %in% names(object)) {
      categorical_vars <- union(categorical_vars, "year")
    }
  }
  
  #Check the requested variables exist
  missing_numeric <- setdiff(numeric_vars, names(object))
  missing_categorical <- setdiff(categorical_vars, names(object))
  
  if (length(missing_numeric) > 0) {
    stop(
      "Numeric variable(s) not found: ",
      paste(missing_numeric, collapse = ", ")
    )
  }
  
  if (length(missing_categorical) > 0) {
    stop(
      "Categorical variable(s) not found: ",
      paste(missing_categorical, collapse = ", ")
    )
  }
  
  weights <- object$PWGTP
  
  #Calculate weighted means and standard deviations
  numeric_summary <- lapply(numeric_vars, function(variable) {
    
    values <- object[[variable]]
    complete <- !is.na(values) & !is.na(weights)
    
    values_complete <- values[complete]
    weights_complete <- weights[complete]
    
    sample_mean <- sum(
      values_complete * weights_complete
    ) / sum(weights_complete)
    
    sample_sd <- sqrt(
      sum(
        values_complete^2 * weights_complete
      ) / sum(weights_complete) -
        sample_mean^2
    )
    
    list(
      mean = sample_mean,
      standard_deviation = sample_sd
    )
  })
  
  names(numeric_summary) <- numeric_vars
  
  # Calculate weighted counts for categorical variables
  categorical_summary <- lapply(categorical_vars, function(variable) {
    
    values <- object[[variable]]
    complete <- !is.na(values) & !is.na(weights)
    
    values_complete <- factor(values[complete])
    weights_complete <- weights[complete]
    
    counts <- tapply(
      weights_complete,
      values_complete,
      sum
    )
    
    return(counts)
  })
  
  names(categorical_summary) <- categorical_vars
  
  #Return the summaries as a named list
  list(
    numeric_summaries = numeric_summary,
    categorical_counts = categorical_summary
  )
}

Graphical summaries

library(ggplot2)

plot.census <- function(x, cat_var, num_var, ...) {
  
  #Both variables need to be specified
  if (missing(cat_var) || missing(num_var)) {
    stop("Specify one categorical variable and one numeric variable.")
  }
  
  #Check the variables exist
  if (!cat_var %in% names(x)) {
    stop(paste(cat_var, "was not found in the census tibble."))
  }
  
  if (!num_var %in% names(x)) {
    stop(paste(num_var, "was not found in the census tibble."))
  }
  
  #Check PWGTP is available
  if (!"PWGTP" %in% names(x)) {
    stop("The census tibble must contain PWGTP.")
  }
  
  #Check that the numeric variable is numeric
  if (!is.numeric(x[[num_var]])) {
    stop(paste(num_var, "must be numeric."))
  }
  ggplot(
    x,
    aes(
      x = factor(get(cat_var)),
      y = get(num_var),
      weight = PWGTP
    )
  ) +
    geom_boxplot(...) +
    labs(
      title = paste(num_var, "by", cat_var),
      x = cat_var,
      y = num_var
    ) +
    theme_minimal()
}

To create the graphical summary, I wrote a plot.census() method for objects with the "census" class. The function requires the user to provide one categorical variable and one numeric variable. It first checks that both variables exist in the data and that the selected numeric variable is actually numeric. It then creates a boxplot showing the distribution of the numeric variable for each level of the categorical variable. I included PWGTP as the weight so that each row contributes according to the number of people it represents. Since this is a method for the "census" class, R automatically uses it when plot() is called on a census object.
